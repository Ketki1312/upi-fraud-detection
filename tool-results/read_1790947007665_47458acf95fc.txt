   118→def load_all_models() -> Dict[str, Any]:
   119→    """Load all trained model pipelines into memory (cached)."""
   120→    global _SHAP_BACKGROUND
   121→    with _MODELS_LOCK:
   122→        if _MODELS.get("_loaded"):
   123→            return _MODELS
   124→        # Transaction models
   125→        for name in ("transaction_xgboost", "transaction_lr", "transaction_rf"):
   126→            p = MODELS_DIR / f"{name}.joblib"
   127→            if p.exists():
   128→                _MODELS[name] = joblib.load(p)
   129→        # NLP models
   130→        nlp_files = {
   131→            "nlp_hi": "nlp_tfidf_lr_hi.joblib",
   132→            "nlp_hinglish": "nlp_tfidf_lr_hinglish.joblib",
   133→            "nlp_en": "nlp_tfidf_lr_en.joblib",
   134→            "nlp_combined": "nlp_tfidf_lr_combined.joblib",
   135→        }
   136→        for name, fname in nlp_files.items():
   137→            p = MODELS_DIR / fname
   138→            if p.exists():
   139→                _MODELS[name] = joblib.load(p)
   140→        # Precomputed SHAP background (small, 100 rows)
   141→        bg_path = MODELS_DIR / "shap_background.joblib"
   142→        if bg_path.exists():
   143→            try:
   144→                _SHAP_BACKGROUND = joblib.load(bg_path)
   145→            except Exception:
   146→                _SHAP_BACKGROUND = None
   147→        _MODELS["_loaded"] = True
   148→    return _MODELS

   186→# ---------------------------------------------------------------------------
   187→# Feature engineering for transactions
   188→# ---------------------------------------------------------------------------
   189→def derive_transaction_features(tx: TransactionInput) -> pd.DataFrame:
   190→    """Build the 14-feature PaySim row from raw input."""
   191→    type_ = (tx.type or "").upper()
   192→    type_map = {
   193→        "CASH_IN": "type_CASH_IN",
   194→        "CASH_OUT": "type_CASH_OUT",
   195→        "DEBIT": "type_DEBIT",
   196→        "PAYMENT": "type_PAYMENT",
   197→        "TRANSFER": "type_TRANSFER",
   198→    }
   199→    feat = {
   200→        "amount": float(tx.amount),
   201→        "oldbalanceOrg": float(tx.oldbalanceOrg),
   202→        "newbalanceOrig": float(tx.newbalanceOrig),
   203→        "oldbalanceDest": float(tx.oldbalanceDest),
   204→        "newbalanceDest": float(tx.newbalanceDest),
   205→    }
   206→    # Derived balance-change features
   207→    feat["orig_balance_change"] = feat["newbalanceOrig"] - feat["oldbalanceOrg"]
   208→    feat["dest_balance_change"] = feat["newbalanceDest"] - feat["oldbalanceDest"]
   209→    feat["amount_to_orig_balance"] = (
   210→        feat["amount"] / feat["oldbalanceOrg"]
   211→        if feat["oldbalanceOrg"] else 0.0
   212→    )
   213→    feat["amount_to_dest_balance"] = (
   214→        feat["amount"] / feat["oldbalanceDest"]
   215→        if feat["oldbalanceDest"] else 0.0
   216→    )
   217→    # Type one-hot
   218→    for col_suffix in ["CASH_IN", "CASH_OUT", "DEBIT", "PAYMENT", "TRANSFER"]:
   219→        feat[f"type_{col_suffix}"] = 1.0 if type_ == col_suffix else 0.0
   220→    # Build dataframe with feature order
   221→    row = {f: feat[f] for f in TX_FEATURES}
   222→    return pd.DataFrame([row], columns=TX_FEATURES)

   225→# ---------------------------------------------------------------------------
   226→# Language detection
   227→# ---------------------------------------------------------------------------
   228→DEVANAGARI_RE = re.compile(r"[\u0900-\u097F]")
   229→HINGLISH_TOKENS = re.compile(
   230→    r"\b(aap|aapka|aapke|karo|kar|hai|hain|nahi|nahin|se|ko|ki|ka|ke|liye|"
   231→    r"rupaye|rupee|paisa|otp|kyc|turant|abhi|jaldi|verify|account|block|"
   232→    r"warning|kaise|kyu|kya|ho|raha|rahe|chahiye|dena|deni|leva|liya|"
   233→    r"payment|refund|bhej|bheji|samjh|smjh|madad|help|"
   234→    r"aapko|mera|meri|tum|tumhara)\b",
   235→    re.IGNORECASE,
   236→)
   237→
   238→
   239→def detect_language(text: str) -> str:
   240→    if not text:
   241→        return "en"
   242→    if DEVANAGARI_RE.search(text):
   243→        return "hi"
   244→    if HINGLISH_TOKENS.search(text):
   245→        return "hinglish"
   246→    return "en"

   297→@app.post("/api/v1/transactions/analyze")
   298→def analyze_transaction(tx: TransactionInput) -> Dict[str, Any]:
   299→    """Run XGBoost fraud prediction + SHAP explanation on a single transaction."""
   300→    if "transaction_xgboost" not in _MODELS:
   301→        raise HTTPException(status_code=503,
   302→                            detail="Transaction model not loaded. Run train.py first.")
   303→    pipeline = _MODELS["transaction_xgboost"]
   304→    X = derive_transaction_features(tx)
   305→    proba = float(pipeline.predict_proba(X)[0, 1])
   306→    pred = int(proba >= 0.5)
   307→    score, level = transaction_risk_score(proba)
   308→
   309→    # SHAP explanation
   310→    tx_meta = get_transaction_metadata()
   311→    # Use precomputed SHAP background (100 legitimate rows saved by train.py)
   312→    background = _SHAP_BACKGROUND
   313→    explainer = TransactionExplainer(
   314→        pipeline, feature_names=TX_FEATURES,
   315→        background=background, method="auto"
   316→    )
   317→    explanation = explainer.explain(X)
   318→
   319→    return {
   320→        "prediction": pred,
   321→        "probability": round(proba, 6),
   322→        "risk_score": round(score, 4),
   323→        "risk_level": level,
   324→        "model_version": tx_meta.get("active_version", "1.0.0"),
   325→        "model_name": "transaction_xgboost",
   326→        "explanation": explanation,
   327→        "input_features": X.iloc[0].to_dict(),
   328→        "disclaimer": DISCLAIMER,
   329→    }

   332→@app.post("/api/v1/messages/analyze")
   333→def analyze_message(msg: MessageInput) -> Dict[str, Any]:
   334→    """Run NLP scam prediction + LIME explanation on a single message."""
   335→    # Auto-detect language if not provided
   336→    language = (msg.language or "").lower()
   337→    if language not in ("hi", "hinglish", "en"):
   338→        language = detect_language(msg.text)
   339→
   340→    model_name_map = {"hi": "nlp_hi", "hinglish": "nlp_hinglish", "en": "nlp_en"}
   341→    model_name = model_name_map.get(language, "nlp_en")
   342→    if model_name not in _MODELS:
   343→        # Fall back to combined
   344→        if "nlp_combined" in _MODELS:
   345→            model_name = "nlp_combined"
   346→        else:
   347→            raise HTTPException(
   348→                status_code=503,
   349→                detail=f"No NLP model available for language={language}; "
   350→                       f"train models first."
   351→            )
   352→    pipeline = _MODELS[model_name]
   353→    text = msg.text
   354→    proba = float(pipeline.predict_proba([text])[0, 1])
   355→    pred = int(proba >= 0.5)
   356→    score, level = message_risk_score(proba)
   357→
   358→    # LIME explanation
   359→    explainer = MessageExplainer(pipeline, class_names=("legitimate", "scam"))
   360→    explanation = explainer.explain(text, num_features=10)
   361→
   362→    nlp_meta = get_nlp_metadata()
   363→    version = nlp_meta.get("models", {}).get(model_name, {}).get("version", "1.0.0")
   364→    return {
   365→        "prediction": pred,
   366→        "probability": round(proba, 6),
   367→        "risk_score": round(score, 4),
   368→        "risk_level": level,
   369→        "model_version": version,
   370→        "model_name": model_name,
   371→        "language_detected": language,
   372→        "explanation": explanation,
   373→        "disclaimer": DISCLAIMER,
   374→    }

   377→@app.post("/api/v1/risk/analyze")
   378→def analyze_risk(req: RiskInput) -> Dict[str, Any]:
   379→    """Fuse transaction + message risks into a unified score."""
   380→    tx_prob = req.transaction_probability
   381→    msg_prob = req.message_probability
   382→    # If pre-computed probabilities are not provided but raw inputs are, compute them
   383→    if tx_prob is None and req.transaction is not None:
   384→        if "transaction_xgboost" in _MODELS:
   385→            X = derive_transaction_features(req.transaction)
   386→            tx_prob = float(_MODELS["transaction_xgboost"].predict_proba(X)[0, 1])
   387→    if msg_prob is None and req.message is not None:
   388→        # Auto-detect language
   389→        lang = (req.message.language or "").lower()
   390→        if lang not in ("hi", "hinglish", "en"):
   391→            lang = detect_language(req.message.text)
   392→        model_name = {"hi": "nlp_hi", "hinglish": "nlp_hinglish",
   393→                      "en": "nlp_en"}.get(lang, "nlp_en")
   394→        if model_name not in _MODELS and "nlp_combined" in _MODELS:
   395→            model_name = "nlp_combined"
   396→        if model_name in _MODELS:
   397→            msg_prob = float(_MODELS[model_name].predict_proba([req.message.text])[0, 1])
   398→    return build_risk_response(tx_prob, msg_prob, method=req.fusion_method)

   531→    """Compute PSI on the validation split as 'current' vs the training
   532→    reference distribution. Real PSI, real distributions."""
   533→    drift_ref = get_drift_reference()
   534→    if not drift_ref:
   535→        raise HTTPException(status_code=503,
   536→                            detail="Drift reference not found. Run train.py first.")
   537→    try:
   538→        val_df = load_paysim_split("validation")
   539→    except Exception as e:
   540→        raise HTTPException(status_code=500, detail=f"Validation split load failed: {e}")
   541→
   542→    # Restrict to features in the reference
   543→    features_in_ref = [k for k in drift_ref.keys() if k not in
   544→                       ("generated_at", "source", "dataset_version",
   545→                        "features", "bins_default")]
   546→    available = [f for f in features_in_ref if f in val_df.columns]
   547→    statuses = evaluate_drift(val_df, drift_ref, features=available)
   548→    return {
   549→        "generated_at": iso_now(),
   550→        "source": "PSI on PaySim validation split vs training reference",
   551→        "features": statuses,
   552→        "thresholds": {
   553→            "watch": float(os.environ.get("DRIFT_WATCH_THRESHOLD", "0.10")),
   554→            "drift": float(os.environ.get("DRIFT_DRIFT_THRESHOLD", "0.25")),
   555→        },
   556→        "disclaimer": DISCLAIMER,
   557→    }

   628→@app.post("/api/v1/models/retrain")
   629→def retrain_endpoint(req: RetrainInput) -> Dict[str, Any]:
   630→    """Trigger candidate retraining + evaluation. Writes outcome to
   631→    metadata/drift_history.json as a retrain event."""
   632→    # Lazy import to avoid circular deps at module load time
   633→    import adaptive
   634→    try:
   635→        prep = adaptive.prepare_candidate_retraining(
   636→            req.model_name, drift_window=req.drift_window
   637→        )
   638→    except Exception as e:
   639→        raise HTTPException(status_code=400, detail=f"prepare_candidate_retraining failed: {e}")
   640→    try:
   641→        outcome = adaptive.evaluate_candidate(req.model_name)
   642→    except FileNotFoundError as e:
   643→        raise HTTPException(status_code=404, detail=str(e))
   644→    except Exception as e:
   645→        raise HTTPException(status_code=500, detail=f"evaluate_candidate failed: {e}")
   646→
   647→    # Persist to drift_history.json (append-only)
   648→    event = {
   649→        "id": f"retrain-{int(time.time())}",
   650→        "type": "retrain",
   651→        "model_name": req.model_name,
   652→        "drift_window": req.drift_window,
   653→        "candidate_version": prep.get("candidate_version"),
   654→        "old_metrics": outcome["old_metrics"],
   655→        "new_metrics": outcome["new_metrics"],
   656→        "comparison": outcome["comparison"],
   657→        "promoted": outcome["promoted"],
   658→        "outcome": outcome["outcome"],
   659→        "created_at": iso_now(),
   660→    }
   661→    adaptive.append_drift_event(event)
   662→
   663→    # Reload models in memory if promoted
   664→    if outcome["promoted"]:
   665→        # Force reload of models from disk
   666→        with _MODELS_LOCK:
   667→            _MODELS.clear()
   668→        load_all_models()
   669→
   670→    return {
   671→        "candidate_version": prep.get("candidate_version"),
   672→        "old_metrics": outcome["old_metrics"],
   673→        "new_metrics": outcome["new_metrics"],
   674→        "promoted": outcome["promoted"],
   675→        "comparison": outcome["comparison"],
   676→        "outcome": outcome["outcome"],
   677→        "event_id": event["id"],
   678→        "created_at": event["created_at"],
   679→        "disclaimer": DISCLAIMER,
   680→    }
   681→
