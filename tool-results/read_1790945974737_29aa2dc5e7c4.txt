   124→def _normalize_text(s: str) -> str:
   125→    """Lowercase, strip ASCII punctuation, but PRESERVE Unicode letters,
   126→    marks (incl. Devanagari matras) and numbers. Replacing `\\W+` would
   127→    strip Devanagari vowel signs (Unicode category Mc) — a real bug for Hindi.
   128→    """
   129→    if not isinstance(s, str):
   130→        return ""
   131→    s = unicodedata.normalize("NFKC", s)
   132→    s = s.lower()
   133→    out_chars = []
   134→    for ch in s:
   135→        cat = unicodedata.category(ch)
   136→        if cat[0] in ("L", "M", "N"):  # letter, mark, number
   137→            out_chars.append(ch)
   138→        else:
   139→            out_chars.append(" ")
   140→    s = "".join(out_chars)
   141→    s = re.sub(r"\s+", " ", s).strip()
   142→    return s

   145→def load_messages_csv(path: Path) -> pd.DataFrame:
   146→    df = pd.read_csv(path)
   147→    df["text"] = df["text"].astype(str).str.strip()
   148→    df["label"] = df["label"].astype(int)
   149→    df["_norm"] = df["text"].map(_normalize_text)
   150→    df["_group"] = df["_norm"]
   151→    # Group IDs by unique normalized text — duplicates share a group id
   152→    unique = df["_group"].unique()
   153→    group_id = {v: i for i, v in enumerate(unique)}
   154→    df["_group_id"] = df["_group"].map(group_id)
   155→    return df

   161→def attempt_download_english_sms() -> Tuple[Optional[pd.DataFrame], Dict[str, Any]]:
   162→    """Attempt to download UCI SMS Spam Collection v1.
   163→    Returns (df or None, info dict). NEVER fabricates data on failure."""
   164→    ENGLISH_DIR.mkdir(parents=True, exist_ok=True)
   165→    info = {
   166→        "url": ENGLISH_DOWNLOAD_URL,
   167→        "attempted_at": iso_now(),
   168→        "success": False,
   169→        "error": None,
   170→    }
   171→    try:
   172→        print(f"[train] attempting English SMS Spam download: {ENGLISH_DOWNLOAD_URL}")
   173→        req = urllib.request.Request(
   174→            ENGLISH_DOWNLOAD_URL,
   175→            headers={"User-Agent": "UPI-Fraud-Research-Prototype/1.0"},
   176→        )
   177→        with urllib.request.urlopen(req, timeout=20) as resp:
   178→            data = resp.read()
   179→        info["bytes_received"] = len(data)
   180→        z = zipfile.ZipFile(io.BytesIO(data))
   181→        # The archive contains a single file: SMSSpamCollection
   182→        names = z.namelist()
   183→        target = None
   184→        for n in names:
   185→            if n.lower().endswith("smsspamcollection") or n == "SMSSpamCollection":
   186→                target = n
   187→                break
   188→        if target is None and len(names) == 1:
   189→            target = names[0]
   190→        if target is None:
   191→            info["error"] = f"could not locate SMSSpamCollection in archive: {names}"
   192→            return None, info
   193→        with z.open(target) as f:
   194→            raw = f.read().decode("latin-1")
   195→        rows = []
   196→        for line in raw.splitlines():
   197→            line = line.strip()
   198→            if not line:
   199→                continue
   200→            if line.startswith(("ham\t", "spam\t")):
   201→                lab, txt = line.split("\t", 1)
   202→                rows.append((lab, txt))
   203→        if not rows:
   204→            info["error"] = "no rows parsed from SMSSpamCollection"
   205→            return None, info
   206→        df = pd.DataFrame(rows, columns=["raw_label", "text"])
   207→        df["label"] = (df["raw_label"] == "spam").astype(int)
   208→        df["language"] = "English"
   209→        df["category"] = df["raw_label"]
   210→        df["source"] = "UCI_SMS_Spam_v1"
   211→        df = df[["text", "language", "category", "label", "source"]]
   212→        info["success"] = True
   213→        info["rows"] = int(len(df))
   214→        info["label_distribution"] = {
   215→            str(k): int(v) for k, v in df["label"].value_counts().items()
   216→        }
   217→        df.to_csv(ENGLISH_DIR / "english_sms_spam.csv", index=False)
   218→        print(f"[train] English SMS Spam downloaded: {len(df):,} rows")
   219→        return df, info
   220→    except urllib.error.URLError as e:
   221→        info["error"] = f"URLError: {e}"
   222→        print(f"[train] English download FAILED (URLError): {e}")
   223→        return None, info
   224→    except Exception as e:
   225→        info["error"] = f"{type(e).__name__}: {e}"
   226→        print(f"[train] English download FAILED: {e}")
   227→        return None, info

   233→def train_logreg_model(X: pd.DataFrame, y: pd.Series) -> Pipeline:
   234→    """LogisticRegression with StandardScaler + class_weight='balanced'."""
   235→    pipe = Pipeline([
   236→        ("scaler", StandardScaler()),

   238→            class_weight="balanced", C=1.0, max_iter=1000,
   239→            random_state=RANDOM_SEED, n_jobs=-1,
   240→        )),
   241→    ])
   242→    pipe.fit(X, y)
   243→    return pipe

   246→def train_rf_model(X: pd.DataFrame, y: pd.Series) -> Pipeline:
   247→    """RandomForestClassifier (no scaling needed for trees)."""
   248→    pipe = Pipeline([
   249→        ("clf", RandomForestClassifier(
   250→            n_estimators=200, max_depth=12,
   251→            class_weight="balanced_subsample",
   252→            n_jobs=-1, random_state=RANDOM_SEED,

   259→def train_xgboost_model(X: pd.DataFrame, y: pd.Series) -> Pipeline:
   260→    """XGBoostClassifier with scale_pos_weight = neg/pos."""
   261→    y = y.astype(int)
   262→    pos = int((y == 1).sum())
   263→    neg = int((y == 0).sum())
   264→    spw = max(neg / max(pos, 1), 1.0)
   265→    pipe = Pipeline([
   266→        ("clf", xgboost.XGBClassifier(
   267→            n_estimators=200, max_depth=6, learning_rate=0.1,
   268→            scale_pos_weight=spw, eval_metric="aucpr",
   269→            random_state=RANDOM_SEED, n_jobs=-1,
   270→            tree_method="hist",
   271→            use_label_encoder=False,
   272→        )),
   273→    ])
   274→    pipe.fit(X, y)
   275→    return pipe

   285→def compute_classification_metrics(pipeline: Pipeline,
   286→                                   X_test: pd.DataFrame,
   287→                                   y_test: pd.Series) -> Dict[str, Any]:
   288→    y_test = y_test.astype(int)
   289→    proba = pipeline.predict_proba(X_test)[:, 1]
   290→    pred = (proba >= 0.5).astype(int)
   291→    metrics: Dict[str, Any] = {
   292→        "precision": float(precision_score(y_test, pred, zero_division=0)),
   293→        "recall": float(recall_score(y_test, pred, zero_division=0)),
   294→        "f1": float(f1_score(y_test, pred, zero_division=0)),
   295→        "roc_auc": float(roc_auc_score(y_test, proba)) if len(np.unique(y_test)) > 1 else 0.0,
   296→        "pr_auc": float(average_precision_score(y_test, proba)) if len(np.unique(y_test)) > 1 else 0.0,
   297→        "confusion_matrix": confusion_matrix(y_test, pred).tolist(),
   298→        "threshold": 0.5,
   299→    }
   300→    return metrics

   306→def train_nlp_pipeline(texts: pd.Series, labels: pd.Series) -> Pipeline:
   307→    """TF-IDF + LogisticRegression. The vectorizer and classifier are wrapped
   308→    in a single sklearn Pipeline that accepts raw text strings."""
   309→    pipe = Pipeline([
   310→        ("tfidf", TfidfVectorizer(
   311→            lowercase=True, ngram_range=(1, 2), max_features=5000,
   312→            min_df=2, sublinear_tf=True, strip_accents="unicode",
   313→        )),
   314→        ("clf", LogisticRegression(
   315→            class_weight="balanced", C=1.0, max_iter=1000,
   316→            random_state=RANDOM_SEED, n_jobs=-1,
   317→        )),
   318→    ])
   319→    pipe.fit(texts.astype(str), labels.astype(int))
   320→    return pipe

   341→def group_aware_split(df: pd.DataFrame,
   342→                       train_frac: float = 0.70,
   343→                       val_frac: float = 0.15,
   344→                       seed: int = RANDOM_SEED) -> Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
   345→    """70/15/15 group-aware split. All rows of a duplicate group go to the
   346→    same split (prevents exact & near-duplicate leakage)."""
   347→    groups = df["_group_id"].values
   348→    unique_groups = np.unique(groups)
   349→    rng = np.random.RandomState(seed)
   350→    rng.shuffle(unique_groups)
   351→    n = len(unique_groups)
   352→    n_train = int(n * train_frac)
   353→    n_val = int(n * val_frac)
   354→    train_g = set(unique_groups[:n_train])
   355→    val_g = set(unique_groups[n_train:n_train + n_val])
   356→    test_g = set(unique_groups[n_train + n_val:])
   357→    train_df = df[df["_group_id"].isin(train_g)].copy()
   358→    val_df = df[df["_group_id"].isin(val_g)].copy()
   359→    test_df = df[df["_group_id"].isin(test_g)].copy()
   360→    return train_df, val_df, test_df

   363→def load_nlp_data_for_language(lang_code: str) -> Tuple[
   364→        Optional[pd.DataFrame], Optional[pd.DataFrame], Optional[pd.DataFrame], Optional[str]]:
   365→    """Return (train_df, test_df, val_df, info_or_None) for a given language.
   366→    Splits are group-aware (duplicate-group aware)."""
   367→    lang_code = (lang_code or "").lower()
   368→    sources: List[Path] = []
   369→    if lang_code == "hi":
   370→        sources = [MESSAGES_DIR / "hindi_messages.csv"]
   371→    elif lang_code == "hinglish":
   372→        sources = [MESSAGES_DIR / "hinglish_messages.csv"]
   373→    elif lang_code == "en":
   374→        p = ENGLISH_DIR / "english_sms_spam.csv"
   375→        if p.exists():
   376→            sources = [p]
   377→        else:
   378→            return None, None, None, "English dataset not available (download failed)"
   379→    elif lang_code == "combined":
   380→        sources = [
   381→            MESSAGES_DIR / "hindi_messages.csv",
   382→            MESSAGES_DIR / "hinglish_messages.csv",
   383→        ]
   384→        p = ENGLISH_DIR / "english_sms_spam.csv"
   385→        if p.exists():
   386→            sources.append(p)
   387→    else:
   388→        return None, None, None, f"unknown language: {lang_code}"
   389→
   390→    frames = []
   391→    for p in sources:
   392→        if not p.exists():
   393→            continue
   394→        df = load_messages_csv(p)
   395→        frames.append(df)
   396→    if not frames:
   397→        return None, None, None, f"no source files for {lang_code}"
   398→    full = pd.concat(frames, ignore_index=True) if len(frames) > 1 else frames[0]
   399→    # Rebuild group ids across the combined frame
   400→    unique = full["_norm"].unique()
   401→    gid = {v: i for i, v in enumerate(unique)}
   402→    full["_group_id"] = full["_norm"].map(gid)
   403→    # Group-aware 70/15/15 split
   404→    train_df, val_df, test_df = group_aware_split(full)
   405→    return train_df, test_df, val_df, None

   411→def build_and_save_drift_reference(train_df: pd.DataFrame) -> Dict[str, Any]:
   412→    ref = build_reference_distribution(train_df, TX_FEATURES, bins=10)
   413→    out = {
   414→        "generated_at": iso_now(),
   415→        "source": "PaySim training split",
   416→        "dataset_version": DATASET_VERSION,
   417→        "features": TX_FEATURES,
   418→        "bins_default": 10,
   419→        **ref,
   420→    }
   421→    (METADATA_DIR / "drift_reference.json").write_text(json.dumps(out, indent=2))
   422→    print(f"[train] saved drift_reference.json ({len(ref)} features)")
   423→
   424→    # Also save a small SHAP background sample (100 legitimate rows) so the
   425→    # FastAPI service does not need to load the full 613k-row train_df for
   426→    # every SHAP request. The KernelExplainer fallback uses this; the
   427→    # TreeExplainer used for XGBoost does not need a background.
   428→    bg = train_df[train_df["isFraud"] == 0][TX_FEATURES].sample(
   429→        n=min(100, len(train_df[train_df["isFraud"] == 0])),
   430→        random_state=RANDOM_SEED,
   431→    )
   432→    joblib.dump(bg, MODELS_DIR / "shap_background.joblib")
   433→    print(f"[train] saved shap_background.joblib ({len(bg)} rows)")
   434→    return out

   437→# ---------------------------------------------------------------------------
   438→# Main training orchestration
   439→# ---------------------------------------------------------------------------
   440→def train_transaction_models() -> Dict[str, Any]:
   441→    """Train LR / RF / XGBoost on PaySim. XGBoost is the active model."""
   442→    print("[train] === Transaction models ===")
   443→    train_df = load_paysim_split("train")
   444→    test_df = load_paysim_split("test")
   445→    X_full = train_df[TX_FEATURES]
   446→    y_full = train_df["isFraud"].astype(int)
   447→    X_test = test_df[TX_FEATURES]
   448→    y_test = test_df["isFraud"].astype(int)
   449→
   450→    pos = int((y_full == 1).sum())
   451→    neg = int((y_full == 0).sum())
   452→    spw = max(neg / max(pos, 1), 1.0)
   453→    print(f"[train] train rows={len(train_df):,} fraud={pos:,} legit={neg:,} "
   454→          f"scale_pos_weight={spw:.2f}")
   455→    print(f"[train] test rows={len(test_df):,} fraud={int((y_test==1).sum()):,}")
   456→
   457→    # ---- LR & RF on a 100k subsample for efficiency (document this) ----
   458→    rng = np.random.RandomState(RANDOM_SEED)
   459→    # Stratified subsample: keep all fraud + 100k legit
   460→    legit_idx = np.where(y_full == 0)[0]
   461→    fraud_idx = np.where(y_full == 1)[0]
   462→    chosen_legit = rng.choice(legit_idx, size=min(100_000, len(legit_idx)),
   463→                              replace=False)
   464→    sub_idx = np.concatenate([chosen_legit, fraud_idx])
   465→    rng.shuffle(sub_idx)
   466→    X_sub = X_full.iloc[sub_idx].reset_index(drop=True)
   467→    y_sub = y_full.iloc[sub_idx].reset_index(drop=True)
   468→    print(f"[train] LR/RF subsample: {len(X_sub):,} rows "
   469→          f"({int((y_sub==1).sum()):,} fraud)")
   470→
   471→    models_meta: Dict[str, Any] = {}
   472→
   473→    # --- Logistic Regression ---
   474→    print("[train] training LogisticRegression (subsampled)...")
   475→    t0 = time.time()
   476→    lr_pipe = train_logreg_model(X_sub, y_sub)
   477→    lr_metrics = compute_classification_metrics(lr_pipe, X_test, y_test)
   478→    joblib.dump(lr_pipe, MODELS_DIR / "transaction_lr.joblib")
   479→    print(f"[train] LR done in {time.time()-t0:.1f}s "
   480→          f"F1={lr_metrics['f1']:.4f} recall={lr_metrics['recall']:.4f} "
   481→          f"prec={lr_metrics['precision']:.4f} roc_auc={lr_metrics['roc_auc']:.4f} "
   482→          f"pr_auc={lr_metrics['pr_auc']:.4f}")
   483→    models_meta["transaction_lr"] = {
   484→        "name": "transaction_lr",
   485→        "type": "logistic_regression",
   486→        "version": "1.0.0",
   487→        "active": False,
   488→        "dataset_version": DATASET_VERSION,
   489→        "training_timestamp": iso_now(),
   490→        "features": TX_FEATURES,
   491→        "hyperparameters": {
   492→            "model": "LogisticRegression",
   493→            "scaler": "StandardScaler",
   494→            "class_weight": "balanced",
   495→            "C": 1.0, "max_iter": 1000,
   496→            "random_state": RANDOM_SEED,
   497→            "subsample_legit_rows": 100_000,
   498→            "subsample_keep_all_fraud": True,
   499→        },
   500→        "metrics": lr_metrics,
   501→        "artifact": "models/transaction_lr.joblib",
   502→    }
   503→

   583→    """Train per-language + combined NLP models."""
   584→    print("[train] === NLP models ===")

   610→                "artifact": None,
   611→            }

   613→
   614→        train_df, test_df, val_df, info = load_nlp_data_for_language(lang_code)
   615→        if train_df is None:
   616→            print(f"[train] {model_name} data unavailable: {info}")
   617→            out_models[model_name] = {
   618→                "name": model_name,
   619→                "type": "tfidf_lr",

   632→        print(f"[train] rows: train={len(train_df):,} val={len(val_df):,} "
