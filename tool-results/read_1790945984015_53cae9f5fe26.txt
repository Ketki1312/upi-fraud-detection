   124→   124→def _normalize_text(s: str) -> str:
   125→   125→    """Lowercase, strip ASCII punctuation, but PRESERVE Unicode letters,
   126→   126→    marks (incl. Devanagari matras) and numbers. Replacing `\\W+` would
   127→   127→    strip Devanagari vowel signs (Unicode category Mc) — a real bug for Hindi.
   128→   128→    """
   129→   129→    if not isinstance(s, str):
   130→   130→        return ""
   131→   131→    s = unicodedata.normalize("NFKC", s)
   132→   132→    s = s.lower()
   133→   133→    out_chars = []
   134→   134→    for ch in s:
   135→   135→        cat = unicodedata.category(ch)
   136→   136→        if cat[0] in ("L", "M", "N"):  # letter, mark, number
   137→   137→            out_chars.append(ch)
   138→   138→        else:
   139→   139→            out_chars.append(" ")
   140→   140→    s = "".join(out_chars)
   141→   141→    s = re.sub(r"\s+", " ", s).strip()
   142→   142→    return s

   145→   145→def load_messages_csv(path: Path) -> pd.DataFrame:
   146→   146→    df = pd.read_csv(path)
   147→   147→    df["text"] = df["text"].astype(str).str.strip()
   148→   148→    df["label"] = df["label"].astype(int)
   149→   149→    df["_norm"] = df["text"].map(_normalize_text)
   150→   150→    df["_group"] = df["_norm"]
   151→   151→    # Group IDs by unique normalized text — duplicates share a group id
   152→   152→    unique = df["_group"].unique()
   153→   153→    group_id = {v: i for i, v in enumerate(unique)}
   154→   154→    df["_group_id"] = df["_group"].map(group_id)
   155→   155→    return df

   160→   160→# ---------------------------------------------------------------------------
   161→   161→def attempt_download_english_sms() -> Tuple[Optional[pd.DataFrame], Dict[str, Any]]:
   162→   162→    """Attempt to download UCI SMS Spam Collection v1.
   163→   163→    Returns (df or None, info dict). NEVER fabricates data on failure."""
   164→   164→    ENGLISH_DIR.mkdir(parents=True, exist_ok=True)
   165→   165→    info = {
   166→   166→        "url": ENGLISH_DOWNLOAD_URL,
   167→   167→        "attempted_at": iso_now(),
   168→   168→        "success": False,
   169→   169→        "error": None,
   170→   170→    }
   171→   171→    try:
   172→   172→        print(f"[train] attempting English SMS Spam download: {ENGLISH_DOWNLOAD_URL}")
   173→   173→        req = urllib.request.Request(
   174→   174→            ENGLISH_DOWNLOAD_URL,
   175→   175→            headers={"User-Agent": "UPI-Fraud-Research-Prototype/1.0"},
   176→   176→        )
   177→   177→        with urllib.request.urlopen(req, timeout=20) as resp:
   178→   178→            data = resp.read()
   179→   179→        info["bytes_received"] = len(data)
   180→   180→        z = zipfile.ZipFile(io.BytesIO(data))
   181→   181→        # The archive contains a single file: SMSSpamCollection
   182→   182→        names = z.namelist()
   183→   183→        target = None
   184→   184→        for n in names:
   185→   185→            if n.lower().endswith("smsspamcollection") or n == "SMSSpamCollection":
   186→   186→                target = n
   187→   187→                break
   188→   188→        if target is None and len(names) == 1:
   189→   189→            target = names[0]
   190→   190→        if target is None:
   191→   191→            info["error"] = f"could not locate SMSSpamCollection in archive: {names}"
   192→   192→            return None, info
   193→   193→        with z.open(target) as f:
   194→   194→            raw = f.read().decode("latin-1")
   195→   195→        rows = []
   196→   196→        for line in raw.splitlines():
   197→   197→            line = line.strip()
   198→   198→            if not line:
   199→   199→                continue
   200→   200→            if line.startswith(("ham\t", "spam\t")):
   201→   201→                lab, txt = line.split("\t", 1)
   202→   202→                rows.append((lab, txt))
   203→   203→        if not rows:
   204→   204→            info["error"] = "no rows parsed from SMSSpamCollection"
   205→   205→            return None, info
   206→   206→        df = pd.DataFrame(rows, columns=["raw_label", "text"])
   207→   207→        df["label"] = (df["raw_label"] == "spam").astype(int)
   208→   208→        df["language"] = "English"
   209→   209→        df["category"] = df["raw_label"]
   210→   210→        df["source"] = "UCI_SMS_Spam_v1"
   211→   211→        df = df[["text", "language", "category", "label", "source"]]
   212→   212→        info["success"] = True
   213→   213→        info["rows"] = int(len(df))
   214→   214→        info["label_distribution"] = {
   215→   215→            str(k): int(v) for k, v in df["label"].value_counts().items()
   216→   216→        }
   217→   217→        df.to_csv(ENGLISH_DIR / "english_sms_spam.csv", index=False)
   218→   218→        print(f"[train] English SMS Spam downloaded: {len(df):,} rows")
   219→   219→        return df, info
   220→   220→    except urllib.error.URLError as e:
   221→   221→        info["error"] = f"URLError: {e}"
   222→   222→        print(f"[train] English download FAILED (URLError): {e}")
   223→   223→        return None, info
   224→   224→    except Exception as e:
   225→   225→        info["error"] = f"{type(e).__name__}: {e}"
   226→   226→        print(f"[train] English download FAILED: {e}")
   227→   227→        return None, info

   259→   259→def train_xgboost_model(X: pd.DataFrame, y: pd.Series) -> Pipeline:
   260→   260→    """XGBoostClassifier with scale_pos_weight = neg/pos."""
   261→   261→    y = y.astype(int)
   262→   262→    pos = int((y == 1).sum())
   263→   263→    neg = int((y == 0).sum())
   264→   264→    spw = max(neg / max(pos, 1), 1.0)
   265→   265→    pipe = Pipeline([
   266→   266→        ("clf", xgboost.XGBClassifier(
   267→   267→            n_estimators=200, max_depth=6, learning_rate=0.1,
   268→   268→            scale_pos_weight=spw, eval_metric="aucpr",
   269→   269→            random_state=RANDOM_SEED, n_jobs=-1,
   270→   270→            tree_method="hist",
   271→   271→            use_label_encoder=False,
   272→   272→        )),
   273→   273→    ])
   274→   274→    pipe.fit(X, y)
   275→   275→    return pipe

   285→   285→def compute_classification_metrics(pipeline: Pipeline,
   286→   286→                                   X_test: pd.DataFrame,
   287→   287→                                   y_test: pd.Series) -> Dict[str, Any]:
   288→   288→    y_test = y_test.astype(int)
   289→   289→    proba = pipeline.predict_proba(X_test)[:, 1]
   290→   290→    pred = (proba >= 0.5).astype(int)
   291→   291→    metrics: Dict[str, Any] = {
   292→   292→        "precision": float(precision_score(y_test, pred, zero_division=0)),
   293→   293→        "recall": float(recall_score(y_test, pred, zero_division=0)),
   294→   294→        "f1": float(f1_score(y_test, pred, zero_division=0)),
   295→   295→        "roc_auc": float(roc_auc_score(y_test, proba)) if len(np.unique(y_test)) > 1 else 0.0,
   296→   296→        "pr_auc": float(average_precision_score(y_test, proba)) if len(np.unique(y_test)) > 1 else 0.0,
   297→   297→        "confusion_matrix": confusion_matrix(y_test, pred).tolist(),
   298→   298→        "threshold": 0.5,
   299→   299→    }
   300→   300→    return metrics

   306→   306→def train_nlp_pipeline(texts: pd.Series, labels: pd.Series) -> Pipeline:
   307→   307→    """TF-IDF + LogisticRegression. The vectorizer and classifier are wrapped

   310→   310→        ("tfidf", TfidfVectorizer(
   311→   311→            lowercase=True, ngram_range=(1, 2), max_features=5000,

   313→   313→        )),

   315→   315→            class_weight="balanced", C=1.0, max_iter=1000,
   316→   316→            random_state=RANDOM_SEED, n_jobs=-1,
   317→   317→        )),
   318→   318→    ])
   319→   319→    pipe.fit(texts.astype(str), labels.astype(int))

   341→   341→def group_aware_split(df: pd.DataFrame,
   342→   342→                       train_frac: float = 0.70,
   343→   343→                       val_frac: float = 0.15,
   344→   344→                       seed: int = RANDOM_SEED) -> Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
   345→   345→    """70/15/15 group-aware split. All rows of a duplicate group go to the
   346→   346→    same split (prevents exact & near-duplicate leakage)."""
   347→   347→    groups = df["_group_id"].values
   348→   348→    unique_groups = np.unique(groups)
   349→   349→    rng = np.random.RandomState(seed)
   350→   350→    rng.shuffle(unique_groups)
   351→   351→    n = len(unique_groups)
   352→   352→    n_train = int(n * train_frac)
   353→   353→    n_val = int(n * val_frac)
   354→   354→    train_g = set(unique_groups[:n_train])
   355→   355→    val_g = set(unique_groups[n_train:n_train + n_val])
   356→   356→    test_g = set(unique_groups[n_train + n_val:])
   357→   357→    train_df = df[df["_group_id"].isin(train_g)].copy()
   358→   358→    val_df = df[df["_group_id"].isin(val_g)].copy()
   359→   359→    test_df = df[df["_group_id"].isin(test_g)].copy()
   360→   360→    return train_df, val_df, test_df

   363→   363→def load_nlp_data_for_language(lang_code: str) -> Tuple[
   364→   364→        Optional[pd.DataFrame], Optional[pd.DataFrame], Optional[pd.DataFrame], Optional[str]]:
   365→   365→    """Return (train_df, test_df, val_df, info_or_None) for a given language.
   366→   366→    Splits are group-aware (duplicate-group aware)."""
   367→   367→    lang_code = (lang_code or "").lower()
   368→   368→    sources: List[Path] = []
   369→   369→    if lang_code == "hi":
   370→   370→        sources = [MESSAGES_DIR / "hindi_messages.csv"]
   371→   371→    elif lang_code == "hinglish":
   372→   372→        sources = [MESSAGES_DIR / "hinglish_messages.csv"]
   373→   373→    elif lang_code == "en":
   374→   374→        p = ENGLISH_DIR / "english_sms_spam.csv"
   375→   375→        if p.exists():
   376→   376→            sources = [p]
   377→   377→        else:
   378→   378→            return None, None, None, "English dataset not available (download failed)"
   379→   379→    elif lang_code == "combined":
   380→   380→        sources = [
   381→   381→            MESSAGES_DIR / "hindi_messages.csv",
   382→   382→            MESSAGES_DIR / "hinglish_messages.csv",
   383→   383→        ]
   384→   384→        p = ENGLISH_DIR / "english_sms_spam.csv"
   385→   385→        if p.exists():
   386→   386→            sources.append(p)
   387→   387→    else:
   388→   388→        return None, None, None, f"unknown language: {lang_code}"
   389→   389→
   390→   390→    frames = []
   391→   391→    for p in sources:
   392→   392→        if not p.exists():
   393→   393→            continue
   394→   394→        df = load_messages_csv(p)
   395→   395→        frames.append(df)
   396→   396→    if not frames:
   397→   397→        return None, None, None, f"no source files for {lang_code}"
   398→   398→    full = pd.concat(frames, ignore_index=True) if len(frames) > 1 else frames[0]
   399→   399→    # Rebuild group ids across the combined frame
   400→   400→    unique = full["_norm"].unique()
   401→   401→    gid = {v: i for i, v in enumerate(unique)}
   402→   402→    full["_group_id"] = full["_norm"].map(gid)
   403→   403→    # Group-aware 70/15/15 split
   404→   404→    train_df, val_df, test_df = group_aware_split(full)
   405→   405→    return train_df, test_df, val_df, None

   411→   411→def build_and_save_drift_reference(train_df: pd.DataFrame) -> Dict[str, Any]:
   412→   412→    ref = build_reference_distribution(train_df, TX_FEATURES, bins=10)
   413→   413→    out = {
   414→   414→        "generated_at": iso_now(),
   415→   415→        "source": "PaySim training split",
   416→   416→        "dataset_version": DATASET_VERSION,
   417→   417→        "features": TX_FEATURES,
   418→   418→        "bins_default": 10,
   419→   419→        **ref,
   420→   420→    }
   421→   421→    (METADATA_DIR / "drift_reference.json").write_text(json.dumps(out, indent=2))
   422→   422→    print(f"[train] saved drift_reference.json ({len(ref)} features)")
   423→   423→
   424→   424→    # Also save a small SHAP background sample (100 legitimate rows) so the
   425→   425→    # FastAPI service does not need to load the full 613k-row train_df for
   426→   426→    # every SHAP request. The KernelExplainer fallback uses this; the
   427→   427→    # TreeExplainer used for XGBoost does not need a background.
   428→   428→    bg = train_df[train_df["isFraud"] == 0][TX_FEATURES].sample(
   429→   429→        n=min(100, len(train_df[train_df["isFraud"] == 0])),
   430→   430→        random_state=RANDOM_SEED,
   431→   431→    )
   432→   432→    joblib.dump(bg, MODELS_DIR / "shap_background.joblib")
   433→   433→    print(f"[train] saved shap_background.joblib ({len(bg)} rows)")
   434→   434→    return out

   459→   459→    # Stratified subsample: keep all fraud + 100k legit
   460→   460→    legit_idx = np.where(y_full == 0)[0]
   461→   461→    fraud_idx = np.where(y_full == 1)[0]

   463→   463→                              replace=False)

   466→   466→    X_sub = X_full.iloc[sub_idx].reset_index(drop=True)
   467→   467→    y_sub = y_full.iloc[sub_idx].reset_index(drop=True)
