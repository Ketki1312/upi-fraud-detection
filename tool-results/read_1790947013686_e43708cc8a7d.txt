   189→   189→def derive_transaction_features(tx: TransactionInput) -> pd.DataFrame:
   190→   190→    """Build the 14-feature PaySim row from raw input."""
   191→   191→    type_ = (tx.type or "").upper()
   192→   192→    type_map = {
   193→   193→        "CASH_IN": "type_CASH_IN",
   194→   194→        "CASH_OUT": "type_CASH_OUT",
   195→   195→        "DEBIT": "type_DEBIT",
   196→   196→        "PAYMENT": "type_PAYMENT",
   197→   197→        "TRANSFER": "type_TRANSFER",
   198→   198→    }
   199→   199→    feat = {
   200→   200→        "amount": float(tx.amount),
   201→   201→        "oldbalanceOrg": float(tx.oldbalanceOrg),
   202→   202→        "newbalanceOrig": float(tx.newbalanceOrig),
   203→   203→        "oldbalanceDest": float(tx.oldbalanceDest),
   204→   204→        "newbalanceDest": float(tx.newbalanceDest),
   205→   205→    }
   206→   206→    # Derived balance-change features
   207→   207→    feat["orig_balance_change"] = feat["newbalanceOrig"] - feat["oldbalanceOrg"]
   208→   208→    feat["dest_balance_change"] = feat["newbalanceDest"] - feat["oldbalanceDest"]
   209→   209→    feat["amount_to_orig_balance"] = (
   210→   210→        feat["amount"] / feat["oldbalanceOrg"]
   211→   211→        if feat["oldbalanceOrg"] else 0.0
   212→   212→    )
   213→   213→    feat["amount_to_dest_balance"] = (
   214→   214→        feat["amount"] / feat["oldbalanceDest"]
   215→   215→        if feat["oldbalanceDest"] else 0.0
   216→   216→    )
   217→   217→    # Type one-hot
   218→   218→    for col_suffix in ["CASH_IN", "CASH_OUT", "DEBIT", "PAYMENT", "TRANSFER"]:
   219→   219→        feat[f"type_{col_suffix}"] = 1.0 if type_ == col_suffix else 0.0
   220→   220→    # Build dataframe with feature order
   221→   221→    row = {f: feat[f] for f in TX_FEATURES}
   222→   222→    return pd.DataFrame([row], columns=TX_FEATURES)

   297→   297→@app.post("/api/v1/transactions/analyze")
   298→   298→def analyze_transaction(tx: TransactionInput) -> Dict[str, Any]:
   299→   299→    """Run XGBoost fraud prediction + SHAP explanation on a single transaction."""
   300→   300→    if "transaction_xgboost" not in _MODELS:
   301→   301→        raise HTTPException(status_code=503,
   302→   302→                            detail="Transaction model not loaded. Run train.py first.")
   303→   303→    pipeline = _MODELS["transaction_xgboost"]
   304→   304→    X = derive_transaction_features(tx)
   305→   305→    proba = float(pipeline.predict_proba(X)[0, 1])
   306→   306→    pred = int(proba >= 0.5)
   307→   307→    score, level = transaction_risk_score(proba)
   308→   308→
   309→   309→    # SHAP explanation
   310→   310→    tx_meta = get_transaction_metadata()
   311→   311→    # Use precomputed SHAP background (100 legitimate rows saved by train.py)
   312→   312→    background = _SHAP_BACKGROUND
   313→   313→    explainer = TransactionExplainer(
   314→   314→        pipeline, feature_names=TX_FEATURES,
   315→   315→        background=background, method="auto"
   316→   316→    )
   317→   317→    explanation = explainer.explain(X)
   318→   318→
   319→   319→    return {
   320→   320→        "prediction": pred,
   321→   321→        "probability": round(proba, 6),
   322→   322→        "risk_score": round(score, 4),
   323→   323→        "risk_level": level,
   324→   324→        "model_version": tx_meta.get("active_version", "1.0.0"),
   325→   325→        "model_name": "transaction_xgboost",
   326→   326→        "explanation": explanation,
   327→   327→        "input_features": X.iloc[0].to_dict(),
   328→   328→        "disclaimer": DISCLAIMER,
   329→   329→    }

   334→   334→    """Run NLP scam prediction + LIME explanation on a single message."""
   335→   335→    # Auto-detect language if not provided
   336→   336→    language = (msg.language or "").lower()
   337→   337→    if language not in ("hi", "hinglish", "en"):
   338→   338→        language = detect_language(msg.text)
   339→   339→
   340→   340→    model_name_map = {"hi": "nlp_hi", "hinglish": "nlp_hinglish", "en": "nlp_en"}
   341→   341→    model_name = model_name_map.get(language, "nlp_en")
   342→   342→    if model_name not in _MODELS:
   343→   343→        # Fall back to combined
   344→   344→        if "nlp_combined" in _MODELS:
   345→   345→            model_name = "nlp_combined"
   346→   346→        else:
   347→   347→            raise HTTPException(
   348→   348→                status_code=503,
   349→   349→                detail=f"No NLP model available for language={language}; "
   350→   350→                       f"train models first."
   351→   351→            )
   352→   352→    pipeline = _MODELS[model_name]
   353→   353→    text = msg.text
   354→   354→    proba = float(pipeline.predict_proba([text])[0, 1])

   368→   368→        "risk_level": level,
   369→   369→        "model_version": version,
   370→   370→        "model_name": model_name,
   371→   371→        "language_detected": language,

   377→   377→@app.post("/api/v1/risk/analyze")
   378→   378→def analyze_risk(req: RiskInput) -> Dict[str, Any]:
   379→   379→    """Fuse transaction + message risks into a unified score."""
   380→   380→    tx_prob = req.transaction_probability
   381→   381→    msg_prob = req.message_probability
   382→   382→    # If pre-computed probabilities are not provided but raw inputs are, compute them
   383→   383→    if tx_prob is None and req.transaction is not None:
   384→   384→        if "transaction_xgboost" in _MODELS:
   385→   385→            X = derive_transaction_features(req.transaction)
   386→   386→            tx_prob = float(_MODELS["transaction_xgboost"].predict_proba(X)[0, 1])
   387→   387→    if msg_prob is None and req.message is not None:
   388→   388→        # Auto-detect language
   389→   389→        lang = (req.message.language or "").lower()
   390→   390→        if lang not in ("hi", "hinglish", "en"):
   391→   391→            lang = detect_language(req.message.text)
   392→   392→        model_name = {"hi": "nlp_hi", "hinglish": "nlp_hinglish",
   393→   393→                      "en": "nlp_en"}.get(lang, "nlp_en")
   394→   394→        if model_name not in _MODELS and "nlp_combined" in _MODELS:
   395→   395→            model_name = "nlp_combined"
   396→   396→        if model_name in _MODELS:
   397→   397→            msg_prob = float(_MODELS[model_name].predict_proba([req.message.text])[0, 1])
   398→   398→    return build_risk_response(tx_prob, msg_prob, method=req.fusion_method)

   531→   531→    """Compute PSI on the validation split as 'current' vs the training
   532→   532→    reference distribution. Real PSI, real distributions."""
   533→   533→    drift_ref = get_drift_reference()
   534→   534→    if not drift_ref:
   535→   535→        raise HTTPException(status_code=503,
   536→   536→                            detail="Drift reference not found. Run train.py first.")
   537→   537→    try:
   538→   538→        val_df = load_paysim_split("validation")
   539→   539→    except Exception as e:
   540→   540→        raise HTTPException(status_code=500, detail=f"Validation split load failed: {e}")
   541→   541→
   542→   542→    # Restrict to features in the reference
   543→   543→    features_in_ref = [k for k in drift_ref.keys() if k not in
   544→   544→                       ("generated_at", "source", "dataset_version",
   545→   545→                        "features", "bins_default")]
   546→   546→    available = [f for f in features_in_ref if f in val_df.columns]
   547→   547→    statuses = evaluate_drift(val_df, drift_ref, features=available)
   548→   548→    return {
   549→   549→        "generated_at": iso_now(),
   550→   550→        "source": "PSI on PaySim validation split vs training reference",
   551→   551→        "features": statuses,
   552→   552→        "thresholds": {
   553→   553→            "watch": float(os.environ.get("DRIFT_WATCH_THRESHOLD", "0.10")),
   554→   554→            "drift": float(os.environ.get("DRIFT_DRIFT_THRESHOLD", "0.25")),

   628→   628→@app.post("/api/v1/models/retrain")
   629→   629→def retrain_endpoint(req: RetrainInput) -> Dict[str, Any]:
   630→   630→    """Trigger candidate retraining + evaluation. Writes outcome to
   631→   631→    metadata/drift_history.json as a retrain event."""
   632→   632→    # Lazy import to avoid circular deps at module load time
   633→   633→    import adaptive
   634→   634→    try:
   635→   635→        prep = adaptive.prepare_candidate_retraining(
   636→   636→            req.model_name, drift_window=req.drift_window
   637→   637→        )
   638→   638→    except Exception as e:
   639→   639→        raise HTTPException(status_code=400, detail=f"prepare_candidate_retraining failed: {e}")
   640→   640→    try:
   641→   641→        outcome = adaptive.evaluate_candidate(req.model_name)
   642→   642→    except FileNotFoundError as e:
   643→   643→        raise HTTPException(status_code=404, detail=str(e))
   644→   644→    except Exception as e:
   645→   645→        raise HTTPException(status_code=500, detail=f"evaluate_candidate failed: {e}")
   646→   646→
   647→   647→    # Persist to drift_history.json (append-only)
   648→   648→    event = {
   649→   649→        "id": f"retrain-{int(time.time())}",
   650→   650→        "type": "retrain",
   651→   651→        "model_name": req.model_name,
   652→   652→        "drift_window": req.drift_window,
   653→   653→        "candidate_version": prep.get("candidate_version"),
   654→   654→        "old_metrics": outcome["old_metrics"],
   655→   655→        "new_metrics": outcome["new_metrics"],
   656→   656→        "comparison": outcome["comparison"],
   657→   657→        "promoted": outcome["promoted"],
   658→   658→        "outcome": outcome["outcome"],
   659→   659→        "created_at": iso_now(),
   660→   660→    }
   661→   661→    adaptive.append_drift_event(event)
   662→   662→
   663→   663→    # Reload models in memory if promoted
   664→   664→    if outcome["promoted"]:
   665→   665→        # Force reload of models from disk
   666→   666→        with _MODELS_LOCK:
   667→   667→            _MODELS.clear()
   668→   668→        load_all_models()
   669→   669→
   670→   670→    return {
   671→   671→        "candidate_version": prep.get("candidate_version"),
   672→   672→        "old_metrics": outcome["old_metrics"],
   673→   673→        "new_metrics": outcome["new_metrics"],
   674→   674→        "promoted": outcome["promoted"],
   675→   675→        "comparison": outcome["comparison"],
   676→   676→        "outcome": outcome["outcome"],
   677→   677→        "event_id": event["id"],
   678→   678→        "created_at": event["created_at"],
   679→   679→        "disclaimer": DISCLAIMER,
   680→   680→    }
